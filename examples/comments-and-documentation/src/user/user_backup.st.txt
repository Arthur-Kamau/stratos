package models;

// Rectangle class implementing Shape
class Rectangle : Shape {
    var width: double;
    var height: double;

    constructor(w: double, h: double) {
        this.width = w;
        this.height = h;
    }

 pub   fn area() double {
        return this.width * this.height;
    }

    pub   fn perimeter() double {
        return 2.0 * (this.width + this.height);
    }

    pub     fn getWidth() double {
        return this.width;
    }
}
